const voiceRanges = VOICE_IDS.map((id) => [f(SEG[id].start), f(SEG[id].end)]);

export const Ep01Collage: React.FC = () => {
  return (
    <AbsoluteFill style={{backgroundColor: '#F5F0E7'}}>
      <Sequence name="SC01 · Hook" from={SCENE.sc1} durationInFrames={SCENE.sc2 + WIPE}>
        <Sc1Hook />
      </Sequence>
      <Sequence name="SC02 · Chỉ nhìn sản phẩm" from={SCENE.sc2} durationInFrames={SC2_LEN}>
        <Sc2Product />
      </Sequence>
      <Sequence name="SC03 · Công thức" from={SCENE.sc3} durationInFrames={SC3_LEN + WIPE}>
        <Sc3Formula />
      </Sequence>
      <Sequence name="SC04–05 · 3 bìa + kết" from={SCENE.sc4} durationInFrames={DURATION - SCENE.sc4}>
        <Sc45Covers />
      </Sequence>
      <Grain opacity={0.08} />

      {/* Giọng đọc */}
      {VOICE_IDS.map((id) => (
        <Sequence key={id} name={`Giọng ${id}`} from={f(SEG[id].start)} layout="none">
          <Audio src={staticFile(`ep01c/voice/${id}.wav`)} volume={1} />
        </Sequence>
      ))}
      {/* SFX tự tổng hợp */}
      {SFX.map(([name, at, vol], i) => (
        <Sequence key={`sfx-${i}`} name={`SFX ${name}`} from={Math.max(0, at)} durationInFrames={30} layout="none">
          <Audio src={sfx(name)} volume={vol} />
        </Sequence>
      ))}
      {/* Nhạc nền (Mixkit, giấy phép Mixkit) — tự hạ khi có lời */}
      <Audio
        src={staticFile('assets/audio/bgm.mp3')}
        trimBefore={48}
        volume={(fr) => {
          const talking = voiceRanges.some(([a, b]) => fr >= a - 4 && fr < b + 4);
          const fadeOut = Math.min(1, Math.max(0, (DURATION - fr) / 30));
          const fadeIn = Math.min(1, fr / 10);
          return (talking ? 0.055 : 0.1) * fadeOut * fadeIn;
        }}
      />
    </AbsoluteFill>
  );
};
